# CTE Function
    CTE stands for Common Table Expression
    it can be used to create a temporary table and use it in the main query
    
    with CTE_name as (select ...)    ----> create the cte function and it can create temporary table
    select * from CTE_name;          ----> use the temporary table in the main query    

    with CTE_name as (select 
    from 
     where 
     group by 
     having 
     order by 
     limit)
    select * from CTE_name;

    multipal cte

# CTE with subquery

with cte12 as(
    select avg(salary) as avg_salary 
    from employee
)
select department, avg(salary) 
from employee
where avg(salary)>(
    select avg_salary
    from cte12
)
